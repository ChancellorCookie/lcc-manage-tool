- Base URL: `https://lcc.ieu.local/api/v2`
- Auth: OAuth2 (client credentials)
- Relevant tags: Rooms, Discovery

### Step 1: Create `src/lib/api.js` — Fetch wrapper

```js
const BASE = '/api'

async function request(path, options = {}) {
  const res = await fetch(BASE + path, {
    headers: { 'Content-Type': 'application/json', ...options.headers },
    ...options
  })
  if (!res.ok) {
    const err = await res.json().catch(() => ({ error: res.statusText }))
    throw new Error(err.error || `HTTP ${res.status}`)
  }
  return res.json()
}

export const api = {
  rooms: {
    list: () => request('/rooms'),
    get: (id) => request(`/rooms/${encodeURIComponent(id)}`),
    create: (data) => request('/rooms', { method: 'POST', body: JSON.stringify(data) }),
    delete: (id) => request(`/rooms/${encodeURIComponent(id)}`, { method: 'DELETE' }),
    patchMeta: (id, data) => request(`/rooms/${encodeURIComponent(id)}/meta`, { method: 'PATCH', body: JSON.stringify(data) }),
  },
  servers: {
    list: () => request('/servers'),
    create: (data) => request('/servers', { method: 'POST', body: JSON.stringify(data) }),
    delete: (id) => request(`/servers/${id}`, { method: 'DELETE' }),
    getCredentials: (id) => request(`/servers/${id}/credentials`),
    putCredentials: (id, data) => request(`/servers/${id}/credentials`, { method: 'PUT', body: JSON.stringify(data) }),
    deleteCredentials: (id) => request(`/servers/${id}/credentials`, { method: 'DELETE' }),
  }
}
```

### Step 2: Create `Rooms.svelte` — List page
